  <script>

    /* ==========================================
       YOUTUBE RSS CONFIGURATION
       
       IMPORTANT:
       Replace ONLY the Channel ID below.

       Example:
       const CHANNEL_ID = "UCxxxxxxxxxxxxxxxxxxxxxx";

       The script automatically:
       1. Creates the YouTube RSS feed URL
       2. Sends it to rss2json
       3. Gets latest videos
       4. Uses only the latest 3
       5. Parses video ID from guid
       6. Creates thumbnails and YouTube links
    ========================================== */

    const CHANNEL_ID = "YOUR_YOUTUBE_CHANNEL_ID";

    const RSS_URL =
      `https://www.youtube.com/feeds/videos.xml?channel_id=${CHANNEL_ID}`;

    const API_URL =
      `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(RSS_URL)}`;

    const videoGrid = document.getElementById("videoGrid");
    const feedStatus = document.getElementById("feedStatus");
    const feedError = document.getElementById("feedError");


    /* ==========================================
       PARSE VIDEO ID FROM GUID
    ========================================== */
    function getVideoIdFromGuid(guid) {

      if (!guid) return null;

      /*
        YouTube RSS normally provides:

        yt:video:VIDEO_ID

        or a guid containing the video URL.
      */

      if (guid.includes("yt:video:")) {
        return guid.split("yt:video:")[1].trim();
      }

      const match = guid.match(
        /(?:v=|youtu\.be\/|\/video\/)([a-zA-Z0-9_-]{11})/
      );

      return match ? match[1] : null;
    }


    /* ==========================================
       FORMAT DATE
    ========================================== */
    function formatDate(dateString) {

      if (!dateString) return "";

      const date = new Date(dateString);

      return date.toLocaleDateString("en-US", {
        year: "numeric",
        month: "short",
        day: "numeric"
      });
    }


    /* ==========================================
       ESCAPE HTML
    ========================================== */
    function escapeHTML(text) {

      const div = document.createElement("div");

      div.textContent = text || "";

      return div.innerHTML;
    }


    /* ==========================================
       LOAD LATEST VIDEOS
    ========================================== */
    async function loadLatestVideos() {

      if (CHANNEL_ID === "YOUR_YOUTUBE_CHANNEL_ID") {

        feedStatus.innerHTML =
          `<span>Setup required:</span> Add your YouTube Channel ID in the JavaScript configuration.`;

        return;
      }

      try {

        const response = await fetch(API_URL);

        if (!response.ok) {
          throw new Error("RSS request failed");
        }

        const data = await response.json();

        if (!data.items || !data.items.length) {
          throw new Error("No videos found");
        }

        /*
          RSS items are sorted by latest videos.
          Slice keeps only latest 3.
        */
        const latestVideos = data.items.slice(0, 3);

        videoGrid.innerHTML = "";

        latestVideos.forEach((video) => {

          const videoId = getVideoIdFromGuid(video.guid);

          if (!videoId) return;

          const title = escapeHTML(video.title);
          const date = formatDate(video.pubDate);

          const thumbnail =
            `https://i.ytimg.com/vi/${videoId}/hqdefault.jpg`;

          const youtubeUrl =
            `https://www.youtube.com/watch?v=${videoId}`;

          const card = document.createElement("article");

          card.className = "video-card reveal";

          card.innerHTML = `
            <a
              href="${youtubeUrl}"
              target="_blank"
              rel="noopener noreferrer"
              aria-label="${title}"
            >
              <div class="video-thumb">

                <img
                  src="${thumbnail}"
                  alt="${title}"
                  loading="lazy"
                  onerror="this.src='https://i.ytimg.com/vi/${videoId}/maxresdefault.jpg'"
                />

                <div class="play-button">
                  <i class="fa-solid fa-play"></i>
                </div>

              </div>

              <div class="video-info">
                <h3>${title}</h3>
                <p>
                  <i class="fa-regular fa-calendar"></i>
                  ${date}
                </p>
              </div>
            </a>
          `;

          videoGrid.appendChild(card);

        });

        feedStatus.innerHTML =
          `<i class="fa-solid fa-circle-check"></i> Latest <span>3 videos</span> loaded automatically.`;

        activateReveal();

      } catch (error) {

        console.error("RSS Feed Error:", error);

        feedStatus.style.display = "none";

        feedError.style.display = "block";
      }
    }


    /* ==========================================
       MOBILE NAVIGATION
    ========================================== */
    const hamburger = document.getElementById("hamburger");
    const navLinks = document.getElementById("navLinks");

    hamburger.addEventListener("click", () => {

      navLinks.classList.toggle("active");

      if (navLinks.classList.contains("active")) {

        icon.classList.remove("fa-bars");
        icon.classList.add("fa-xmark");

      } else {

        icon.classList.remove("fa-xmark");
        icon.classList.add("fa-bars");

      }

    });

      });

    });


    /* ==========================================
       SCROLL REVEAL ANIMATION
    ========================================== */
    function activateReveal() {

      const revealElements =
        document.querySelectorAll(".reveal");

      const observer = new IntersectionObserver(
        (entries) => {

          entries.forEach((entry) => {

            if (entry.isIntersecting) {

              entry.target.classList.add("active");

              observer.unobserve(entry.target);

            }

          });

        },
        {
          threshold: 0.12
        }
      );

      revealElements.forEach((element) => {
        observer.observe(element);
      });
    }
